<script>
    const TARGET_GOAL = 4800;
    const TOTAL_DAYS = 30;
    let plan = [];
    let completedDays = [];

    // 初始化計劃
    function init() {
        const savedPlan = localStorage.getItem('savings_plan');
        const savedProgress = localStorage.getItem('savings_progress');

        if (savedPlan) {
            plan = JSON.parse(savedPlan);
            completedDays = JSON.parse(savedProgress) || [];
        } else {
            generateNewPlan();
        }
        render();
    }

    function generateNewPlan() {
        plan = [];
        // 10天輕鬆(50-100), 10天中等(120-180), 10天挑戰(200-250)
        for(let i=0; i<10; i++) plan.push(Math.floor(Math.random() * 51) + 50);
        for(let i=0; i<10; i++) plan.push(Math.floor(Math.random() * 61) + 120);
        for(let i=0; i<10; i++) plan.push(Math.floor(Math.random() * 51) + 200);
        
        // 洗牌
        plan.sort(() => Math.random() - 0.5);
        
        // 微調總額
        let currentTotal = plan.reduce((a, b) => a + b, 0);
        plan[29] += (TARGET_GOAL - currentTotal);
        
        localStorage.setItem('savings_plan', JSON.stringify(plan));
        completedDays = [];
        localStorage.setItem('savings_progress', JSON.stringify(completedDays));
    }

    function render() {
        const currentDayIndex = completedDays.length;
        const isFinished = currentDayIndex >= TOTAL_DAYS;

        // 更新金額顯示
        const todayAmount = isFinished ? 0 : plan[currentDayIndex];
        document.getElementById('today-amount').innerText = `$${todayAmount}`;
        
        // 更新進度條
        const totalSaved = completedDays.reduce((sum, idx) => sum + plan[idx], 0);
        document.getElementById('current-total').innerText = `$${totalSaved}`;
        const progressPercent = (totalSaved / TARGET_GOAL) * 100;
        document.getElementById('progress-bar-fill').style.width = `${Math.min(progressPercent, 100)}%`;
        
        document.getElementById('days-left').innerText = `剩餘天數：${TOTAL_DAYS - completedDays.length} 天`;

        // 按鈕狀態
        const btn = document.getElementById('check-btn');
        if(isFinished) {
            btn.innerText = "本月已圓滿達成！";
            btn.disabled = true;
            document.getElementById('status-text').innerText = "太棒了！";
        }

        // 生成歷史網格
        const grid = document.getElementById('history-grid');
        grid.innerHTML = '';
        for(let i=0; i<TOTAL_DAYS; i++) {
            const dot = document.createElement('div');
            dot.className = `day-dot ${i < completedDays.length ? 'done' : ''}`;
            dot.innerText = i + 1;
            grid.appendChild(dot);
        }
    }

    function markAsDone() {
        if(completedDays.length < TOTAL_DAYS) {
            completedDays.push(completedDays.length);
            localStorage.setItem('savings_progress', JSON.stringify(completedDays));
            render();
            alert("成功！記得把錢存入儲蓄戶口喔！💪");
        }
    }

    function resetPlan() {
        if(confirm("確定要重新開始新的 30 天計劃嗎？目前的進度會被清除喔！")) {
            generateNewPlan();
            render();
        }
    }

    init();
</script>
